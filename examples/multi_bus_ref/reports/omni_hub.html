 213       // 2. Ethernet Cable Hot-Plug + Auto-Negotiation + EEE Sleep
 214       eth_cap_d <= eth_peer_cap;
 215       if (!eth_cable_present) begin
 216         // Hot-plug cable unplug -> immediate link drop & abort!

 218       end else begin

 220           E_DOWN: begin eth_st <= E_TRAIN; eth_cnt <= 0; end

 224               else if (eth_peer_cap == 2'b10) begin eth_st <= E_UP;   eth_speed <= 2'b01; end
 225               else                            begin eth_st <= E_FAIL; eth_speed <= 2'b00; end
 226             end else eth_cnt <= eth_cnt + 1;
 227           end
 228           E_UP, E_FAIL: begin

 230           end
 231         endcase
 232         if (eee_sleep_req && eth_st == E_UP && !tx_act && !eth_rx_dv) eee_sleep_ack <= 1;
 233         else if (!eee_sleep_req) eee_sleep_ack <= 0;
 234       end

 236       // 3. PCIe Surprise Hot-Plug + LTSSM + L1 Sleep
 237       if (!pcie_hotplug_present) begin
 238         // Surprise removal -> immediate drop to DETECT!
 239         ltssm_state <= P_DETECT; pcie_cnt <= 0;
 240       end else begin
 241         case (ltssm_state)
 242           P_DETECT: begin ltssm_state <= P_POLL; pcie_cnt <= 0; end
 243           P_POLL:   if (pcie_cnt == 1) begin ltssm_state <= P_CFG; pcie_cnt <= 0; end else pcie_cnt <= pcie_cnt + 1;
 244           P_CFG:    if (pcie_deskew_ok) ltssm_state <= P_L0; else if (pcie_cnt == 2) ltssm_state <= P_FAIL; else pcie_cnt <= pcie_cnt + 1;
 245           P_L0:     if (pm_l1_req && !pcie_buf_act && !pcie_tx_tvalid) ltssm_state <= P_L1;
 246           P_L1:     if (!pm_l1_req) ltssm_state <= P_L0;
 247           P_FAIL:   if (pcie_deskew_ok) begin ltssm_state <= P_DETECT; pcie_cnt <= 0; end
 248           default:  ltssm_state <= P_DETECT;
 249         endcase
 250       end

 252       // 4. IF1: SPI Slave -> Shared MEM (Mode 0/1/2/3)

 255         spi_bcnt <= 0;

 261         end else if (spi_bcnt == 15) begin

 263           fr = {spi_rx[14:0], s_spi_mosi};

 265           spi_bcnt <= 0;
 266         end else begin
 267           if (spi_bcnt > 7) spi_tx <= {spi_tx[6:0], 1'b0};
 268           spi_bcnt <= spi_bcnt + 1;
 269         end
 270       end

 272       // 5. IF2: APB4 Slave (with BUF_CYCS buffer) -> Shared MEM
 273       s_apb_pready <= 0; s_apb_pslverr <= 0;
 274       if (apb_done) begin
 275         apb_done <= 0;
 276       end else if (!apb_buf_act && s_apb_psel && s_apb_penable) begin
 277         apb_wr_r <= s_apb_pwrite; apb_addr_r <= s_apb_paddr; apb_wdata_r <= s_apb_pwdata; apb_strb_r <= s_apb_pstrb;
 278         if (BUF_CYCS == 0) begin
 279           s_apb_pready <= 1; apb_done <= 1;
 280           if (!ddr_ready || s_apb_paddr >= 16'h0100) begin s_apb_pslverr <= 1; s_apb_prdata <= 0; end

 283           end else begin
 284             s_apb_prdata <= {mem[s_apb_paddr[7:0]+8'd3], mem[s_apb_paddr[7:0]+8'd2], mem[s_apb_paddr[7:0]+8'd1], mem[s_apb_paddr[7:0]+8'd0]};
 285           end
 286         end else begin
 287           apb_buf_act <= 1; apb_bcnt <= 4'(BUF_CYCS);
 288         end
 289       end else if (apb_buf_act) begin
 290         if (apb_bcnt <= 1) begin
 291           apb_buf_act <= 0; s_apb_pready <= 1; apb_done <= 1;
 292           if (!ddr_ready || apb_addr_r >= 16'h0100) begin s_apb_pslverr <= 1; s_apb_prdata <= 0; end
 293           else if (apb_wr_r) begin
 294             for (int b = 0; b < 4; b++) if (apb_strb_r[b]) mem[apb_addr_r[7:0] + 8'(b)] <= apb_wdata_r[b*8 +: 8];
 295           end else begin
 296             s_apb_prdata <= {mem[apb_addr_r[7:0]+8'd3], mem[apb_addr_r[7:0]+8'd2], mem[apb_addr_r[7:0]+8'd1], mem[apb_addr_r[7:0]+8'd0]};
 297           end
 298         end else apb_bcnt <= apb_bcnt - 1;
 299       end

 327             s_axi_bresp <= 2'b00;
 328           end else if (axi_waddr_r == 16'h0100) begin

 332         end else axi_wbcnt <= axi_wbcnt - 1;
 333       end

 352           end
 353         end else axi_rbcnt <= axi_rbcnt - 1;
 354       end

 356       // 7. IF4: Ethernet RX -> CRC -> Shared MEM + TX serializer
 357       rx_crc_err <= 0; rx_drop_nolink <= 0;

 359         if (!eth_link_up || !ddr_ready) begin

 363             3'd0: if (eth_rx_data == 8'hD5) begin rx_idx <= 1; rx_xor <= 8'hD5; end

 369             3'd6: begin
 370               if (eth_rx_data == rx_xor) begin
 371                 mem[rx_addr + 8'd0] <= rx_buf[0]; mem[rx_addr + 8'd1] <= rx_buf[1];
 372                 mem[rx_addr + 8'd2] <= rx_buf[2]; mem[rx_addr + 8'd3] <= rx_buf[3];
 373               end else rx_crc_err <= 1;
 374               rx_idx <= 0;
 375             end
 376             default: rx_idx <= 0;
 377           endcase
 378         end
 379       end else rx_idx <= 0;
 380 
 381       eth_tx_en <= 0;
 382       if (tx_act) begin
 383         if (!eth_link_up) begin tx_act <= 0; tx_idx <= 0; end
 384         else begin
 385           eth_tx_en <= 1;

 390             3'd3: begin eth_tx_data <= mem[tx_addr+8'd1]; tx_xor <= tx_xor ^ mem[tx_addr+8'd1]; tx_idx <= 4; end
 391             3'd4: begin eth_tx_data <= mem[tx_addr+8'd2]; tx_xor <= tx_xor ^ mem[tx_addr+8'd2]; tx_idx <= 5; end

 395           endcase
 396         end
 397       end
 398 
 399       // 8. IF5: PCIe TLP (with BUF_CYCS buffer) -> Shared MEM
 400       tlp_reject_nol0 <= 0;
 401       if (pcie_tx_tvalid && pcie_tx_tready) begin pcie_tx_tvalid <= 0; pcie_tx_tlast <= 0; end

 403         if (!pcie_link_up || !ddr_ready) begin
 404           tlp_reject_nol0 <= 1;

 409             mem[pcie_rx_tdata[39:32]+8'd2] <= pcie_rx_tdata[23:16];

 411           end else if (pcie_rx_tdata[63:56] == 8'h00) begin

 416           end
 417         end else begin
 418           pcie_buf_act <= 1; pcie_bcnt <= 4'(BUF_CYCS); pcie_lat <= pcie_rx_tdata;
 419         end
 420       end else if (pcie_buf_act) begin
 421         if (pcie_bcnt <= 1) begin
 422           pcie_buf_act <= 0;
 423           if (!pcie_link_up || !ddr_ready) begin
 424             tlp_reject_nol0 <= 1;
